import React, { useEffect, useRef, useState } from "react";

export default function UnaPequenaCoincidencia() {
  const canvasRef = useRef(null);
  const audioRef = useRef(null); // Referencia para el audio
  const [gameOver, setGameOver] = useState(false);
  const [level, setLevel] = useState(1);
  const [hearts, setHearts] = useState(0);

  const player = useRef({ x: 40, y: 120, vy: 0, onGround: true });
  const partner = useRef({ x: 70, y: 120 });

  const obstacles = useRef([{ x: 300, w: 20, h: 40 }]);
  const coins = useRef([{ x: 200, y: 100 }]);

  useEffect(() => {
    const canvas = canvasRef.current;
    const ctx = canvas.getContext("2d");
    const gravity = 0.6;

    // Reproducir música al iniciar
    if (audioRef.current) {
      audioRef.current.play().catch(err => console.log("Audio no pudo reproducirse:", err));
    }

    function update() {
      if (gameOver) {
        if (audioRef.current) audioRef.current.pause(); // Pausar música en game over
        return;
      }

      // Física jugador
      player.current.vy += gravity;
      player.current.y += player.current.vy;

      if (player.current.y >= 120) {
        player.current.y = 120;
        player.current.vy = 0;
        player.current.onGround = true;
      }

      // Obstáculos
      obstacles.current.forEach(o => (o.x -= 2 + level));

      // Monedas corazón
      coins.current.forEach(c => (c.x -= 2 + level));

      // Colisión con obstáculos
      obstacles.current.forEach(o => {
        if (
          player.current.x < o.x + o.w &&
          player.current.x + 20 > o.x &&
          player.current.y < 140 &&
          player.current.y + 20 > 140 - o.h
        ) {
          setGameOver(true);
        }
      });

      // Colisión con corazones
      coins.current = coins.current.filter(c => {
        if (
          player.current.x < c.x + 10 &&
          player.current.x + 20 > c.x &&
          player.current.y < c.y + 10 &&
          player.current.y + 20 > c.y
        ) {
          setHearts(h => h + 1);
          return false;
        }
        return true;
      });

      // Reset obstáculos y subir nivel
      if (obstacles.current[0].x < -20) {
        obstacles.current.shift();
        obstacles.current.push({ x: 300, w: 20, h: 40 });
        coins.current.push({ x: 260, y: 100 });
        setLevel(l => l + 1);
      }
    }

    function draw() {
      ctx.clearRect(0, 0, 300, 150);

      // Suelo
      ctx.fillStyle = "#444";
      ctx.fillRect(0, 140, 300, 10);

      // Tu personaje (piel blanca, vestido rosado, pelo marrón)
      ctx.fillStyle = "#f5c6d6";
      ctx.fillRect(player.current.x, player.current.y, 20, 20);
      ctx.fillStyle = "#8b5a2b";
      ctx.fillRect(player.current.x, player.current.y - 6, 20, 6);

      // Pareja (trigueña, pelo negro)
      ctx.fillStyle = "#d2a679";
      ctx.fillRect(partner.current.x, partner.current.y, 20, 20);
      ctx.fillStyle = "#000";
      ctx.fillRect(partner.current.x, partner.current.y - 6, 20, 6);

      // Obstáculos
      ctx.fillStyle = "#000";
      obstacles.current.forEach(o => ctx.fillRect(o.x, 140 - o.h, o.w, o.h));

      // Corazones dorados
      ctx.fillStyle = "gold";
      coins.current.forEach(c => ctx.fillRect(c.x, c.y, 10, 10));

      // HUD
      ctx.fillStyle = "black";
      ctx.font = "12px Arial";
      ctx.fillText(`Nivel: ${level}`, 10, 10);
      ctx.fillText(`❤ ${hearts}`, 220, 10);
    }

    function loop() {
      update();
      draw();
      requestAnimationFrame(loop);
    }

    loop();

    // Limpiar al desmontar
    return () => {
      if (audioRef.current) audioRef.current.pause();
    };
  }, [gameOver, level]);

  function jump() {
    if (player.current.onGround) {
      player.current.vy = -10;
      player.current.onGround = false;
    }
  }

  function restart() {
    setGameOver(false);
    setLevel(1);
    setHearts(0);
    player.current = { x: 40, y: 120, vy: 0, onGround: true };
    obstacles.current = [{ x: 300, w: 20, h: 40 }];
    coins.current = [{ x: 200, y: 100 }];
    if (audioRef.current) audioRef.current.play();
  }

  return (
    <div className="flex flex-col items-center gap-2">
      <h1 className="text-xl font-bold">Una pequeña coincidencia 💖</h1>
      <p>Niveles y corazones dorados</p>
      <canvas ref={canvasRef} width={300} height={150} className="border" />
      {gameOver && (
        <div className="text-center">
          <p className="text-red-600">Game Over 💔</p>
          <button onClick={restart} className="px-4 py-1 bg-green-500 text-white rounded">Reiniciar</button>
        </div>
      )}
      <button onClick={jump} className="px-4 py-1 bg-pink-500 text-white rounded">Saltar</button>
      {/* Elemento de audio oculto */}
      <audio ref={audioRef} loop src="/i-can-be-needy.mp3" style={{ display: 'none' }} />
      <p className="text-xs">🎵 Música: I Can Be Needy (Ariana Grande) – reproducida en bucle como fondo</p>
    </div>
  );
}
